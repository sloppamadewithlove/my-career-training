"""Offline research diagnostic: measure real RLBot render payloads for MCT.

This is a research tool, not a product feature and not a test. It never
connects a socket, starts an RLBot server, launches Rocket League or any other
game/server process, or touches the network. One caveat, stated precisely:
importing the installed ``rlbot`` package runs ``rlbot.utils.logging``, whose
``get_logger`` calls ``check_color`` on Windows, which issues the harmless
console-color shell command ``os.system("color")``. That dependency behavior is
neither suppressed nor monkeypatched here; it is recorded in the report's
limitations.
It drives the actually installed ``rlbot`` ``Renderer`` and the real
``SocketRelay.send_msg``/``send_bytes`` code paths against a fake in-memory
socket, and the unchanged ``GameSession.render``/``draw_target``/
``draw_ghost_actions`` methods bound to a synthetic session built with
``GameSession.__new__`` (no ``__init__``, no thread), so the measured payload
bytes and RenderMessage counts come from production code.

Scope of the claim: the *client* pipeline only (Python serialization, the 2-byte
big-endian length prefix, and the client-side oversized rejection). It says
nothing about the RLBot server, the closed-source Bridge game-tick byte budget,
or game-side draw persistence.

Output is one sanitized JSON document on stdout (relative source labels, no
absolute paths, no credentials). No files are written by this script; bytecode
writing is disabled before the imports so importing the project does not create
``__pycache__`` entries either. Exit code is non-zero if any check fails.

Run only after parent review, offline, with the project interpreter.
"""

def check(name: str, ok: bool, detail=None) -> bool:
    CHECKS.append({"name": name, "ok": bool(ok), "detail": detail})
    return bool(ok)


def rel(path) -> str:
    """Relative source label for repo files; basename only for installed files."""
    resolved = Path(path).resolve()
    try:
        return resolved.relative_to(REPO_ROOT).as_posix()
    except ValueError:
        return resolved.name


def sha256_of(path) -> str:
    try:
        return hashlib.sha256(Path(path).read_bytes()).hexdigest()
    except OSError:
        return "unavailable"


def module_source(module) -> dict:
    path = getattr(module, "__file__", None)
    if not path:
        return {"label": "unavailable", "sha256": "unavailable", "bytes": None,
                "outside_repo": None}
    resolved = Path(path).resolve()
    return {
        "label": rel(resolved),
        "sha256": sha256_of(resolved),
        "bytes": resolved.stat().st_size if resolved.exists() else None,
        "outside_repo": REPO_ROOT not in resolved.parents,
    }

class FakeSocket:
    """In-memory stand-in for ``socket.socket``; never performs I/O."""

    def __init__(self):
        self.sent: list[bytes] = []
        self.connect_calls = 0
        self.recv_calls = 0
        self.timeout_calls = 0
        self.setsockopt_calls = 0
        self.shutdown_calls = 0
        self.close_calls = 0

    def sendall(self, data):
        self.sent.append(bytes(data))

    def connect(self, address):  # pragma: no cover - must never run
        self.connect_calls += 1
        raise AssertionError("diagnostic must not connect a socket")

    def recv(self, size):  # pragma: no cover - must never run
        self.recv_calls += 1
        raise AssertionError("diagnostic must not read from a socket")

    def recv_into(self, view):  # pragma: no cover - must never run
        self.recv_calls += 1
        raise AssertionError("diagnostic must not read from a socket")

    def settimeout(self, value):
        self.timeout_calls += 1

    def setsockopt(self, *args):
        self.setsockopt_calls += 1

    def setblocking(self, flag):
        return None

    def shutdown(self, how):
        self.shutdown_calls += 1

    def close(self):
        # Harmless: SocketRelay.__del__ may call this.
        self.close_calls += 1

    def getsockname(self):
        return ("127.0.0.1", 0)


class RecordingHandler(logging.Handler):
    def __init__(self):
        super().__init__(level=logging.DEBUG)
        self.records: list[dict] = []

    def emit(self, record):
        self.records.append({"level": record.levelname, "message": record.getMessage()})


def make_relay(agent_id: str) -> tuple[SocketRelay, FakeSocket, RecordingHandler]:
    """Build a SocketRelay wired to a fake socket without calling connect()."""
    handler = RecordingHandler()
    logger = logging.getLogger(f"probe.{agent_id}")
    logger.handlers = [handler]
    logger.setLevel(logging.DEBUG)
    logger.propagate = False
    relay = SocketRelay(agent_id, logger=logger)
    socket = FakeSocket()
    relay.socket = socket
    # Set directly; SocketRelay.connect() is never called.
    relay.is_connected = True
    relay.can_render = True
    relay._running = False
    return relay, socket, handler


def make_capturing_relay(agent_id: str):
    """Relay whose ``send_msg`` is a recording wrapper around the real one.

    Instrumentation, stated precisely: the wrapper re-serializes the *same*
    message object with the same ``flat.InterfacePacket(msg).pack()`` call that
    ``SocketRelay.send_msg`` performs, records those bytes, and then delegates to
    the unchanged ``SocketRelay.send_msg``. The recorded bytes therefore come
    from the production serialization path, not from a hand-built reconstruction.
    For accepted sends the recorded bytes are asserted equal to the bytes the
    fake socket received, which proves the delegation really happened.
    """
    relay, socket, handler = make_relay(agent_id)
    captured: list[dict] = []

    def wrapper(msg):
        payload = flat.InterfacePacket(msg).pack()
        captured.append({"kind": type(msg).__name__, "payload": payload})
        SocketRelay.send_msg(relay, msg)

    relay.send_msg = wrapper
    return relay, socket, handler, captured, wrapper


def decode_frames(socket: FakeSocket) -> list[dict]:
    frames = []
    for raw in socket.sent:
        prefix = raw[:2]
        size = int.from_bytes(prefix, "big")
        payload = raw[2:]
        frames.append(
            {
                "prefix_hex": prefix.hex(),
                "prefix_value": size,
                "payload_bytes": len(payload),
                "payload": payload,
                "prefix_matches_payload": size == len(payload),
            }
        )
    return frames


RENDER_VARIETIES = {"Line3D", "PolyLine3D", "Rect2D", "Rect3D", "String2D", "String3D"}


def describe_payload(payload: bytes) -> dict:
    """Decode one serialized InterfacePacket payload for description only.

    Byte counts always come from the raw frame; this only names the message
    kinds. A decode failure is reported as ``unparsed`` and fails a check.
    """
    try:
        packet = flat.InterfacePacket.unpack(payload)
        message = packet.message
    except Exception as error:  # noqa: BLE001 - reported, not hidden
        return {
            "kind": "unparsed",
            "group_id": None,
            "message_counts": {},
            "total_messages": 0,
            "text_utf8_bytes": 0,
            "polyline_points": 0,
            "decode_error": f"{type(error).__name__}: {error}",
        }
    if isinstance(message, flat.RenderGroup):
        counts: dict[str, int] = {}
        text_bytes = 0
        polyline_points = 0
        render_messages = message.render_messages
        for render_message in render_messages:
            variety = render_message.variety
            name = type(variety).__name__
            counts[name] = counts.get(name, 0) + 1
            if isinstance(variety, flat.String2D):
                text_bytes += len(variety.text.encode("utf-8"))
            elif isinstance(variety, flat.PolyLine3D):
                polyline_points += len(variety.points)
        return {
            "kind": "RenderGroup",
            "group_id": message.id,
            "message_counts": counts,
            "total_messages": len(render_messages),
            "text_utf8_bytes": text_bytes,
            "polyline_points": polyline_points,
        }
    return {
        "kind": type(message).__name__,
        "group_id": getattr(message, "id", None),
        "message_counts": {},
        "total_messages": 0,
        "text_utf8_bytes": 0,
        "polyline_points": 0,
    }

class FakeEngine:
    """Minimal stand-in for the training engine used by GameSession.render.

    Not live and not the full engine: it only supplies the attributes and the
    ``summary()`` shape that ``render`` reads.
    """

    def __init__(self, lesson, preferences, *, state="active", started_at=0.0,
                 finished=False, coaching=None):
        self.lesson = lesson
        self.preferences = preferences
        self.state = state
        self.started_at = started_at
        self.finished = finished
        self.coaching = dict(coaching or {})

    def summary(self):
        return {
            "coaching": dict(self.coaching),
            "duration": SYNTHETIC_DURATION,
            "feedback": SYNTHETIC_FEEDBACK,
            "id": "synthetic-attempt",
        }

    if with_controller:

# Attributes ``GameSession.render``/``draw_target``/``draw_ghost_actions`` read.
# Verified against game_worker.py:1406-1515: render touches cancel_start,
# stopping, renderer, engine, request, practice_snapshot, practice,
# legacy_playback, coach_snapshot; the two draw helpers touch only renderer.
# This is the explicit seed list only; it is not a general guard against a
# future render() reading an attribute that is not listed here.
RENDER_ATTRIBUTES = (
    "cancel_start", "stopping", "renderer", "engine", "request",
    "practice_snapshot", "practice", "legacy_playback", "coach_snapshot",
)


def build_session(lesson, preferences, *, native_hud, practice, coach, engine):
    """Bind the real render methods to a synthetic session.

    One code path only: ``GameSession.__new__`` plus explicit attribute seeding.
    ``__init__`` is never called, so no thread, no ``session_uuid``, no
    ``.review`` import and no other constructor side effect runs.
    """
    session = GameSession.__new__(GameSession)
    session.request = {
        "session_id": "00000000-0000-4000-8000-000000000000",
        "lesson_id": lesson["id"],
        "native_hud": native_hud,
    }
    session.cancel_start = threading.Event()
    session.stopping = False
    session.renderer = None
    session.engine = engine
    session.practice_snapshot = practice
    # render only tests ``self.practice is None``; the snapshot dict is the
    # synthetic stand-in for the practice controller object.
    session.practice = practice if practice.get("visible") else None
    session.legacy_playback = False
    session.coach_snapshot = coach
    return session


def session_attributes_present(session) -> dict:
    """Verify the seeded attributes are real instance fields, not class descriptors."""
    return {
        name: (name in vars(session), type(getattr(session, name, None)).__name__)
        for name in RENDER_ATTRIBUTES
    }


def expected_target_lines(targets) -> int:
    total = 0
    for target in targets:
        kind = target.get("type")
        if kind == "gate":
            total += TARGET_LINE_COUNTS["gate"]
        elif kind == "goal":
            total += TARGET_LINE_COUNTS["goal"]
        elif kind == "plane" and target.get("axis") == 1:
            total += TARGET_LINE_COUNTS["plane_axis_1"]
        # carry / ball_reset / recovery / other kinds match no draw_target branch.
    return total


def expected_string_count(*, steps, preferences, native_hud, practice_visible,
                          coach_visible, engine_state, coaching) -> int:
    """Mirror of the text branches in game_worker.py:1419-1490."""
    count = 0
    if preferences["explanations"] and not native_hud:
        count += 1 + min(6, steps)  # title + steps
    if preferences["actions"] and not native_hud:
        count += 1
    if preferences["timing"] and not native_hud:
        count += 1
    message = coaching.get("message")
    interrupted = engine_state == "interrupted"
    if interrupted:
        message = coaching.get("interruption_reason") or SYNTHETIC_FEEDBACK or message
    if not native_hud and message and (preferences["explanations"] or interrupted):
        count += 1
    stages = coaching.get("stages") or []
    next_stage = coaching.get("next_stage")
    if (preferences["explanations"] and not native_hud and isinstance(next_stage, int)
            and 0 <= next_stage < len(stages)):
        count += 1
    if practice_visible:
        count += 1  # "GHOST · label"
        if not native_hud:
            count += 2  # draw_ghost_actions: inputs line + provenance line
    elif coach_visible and preferences["ghost"]:
        count += 1  # coach label
    return count


def run_fixture(label, lesson, preferences, *, native_hud=False, practice=None,
                coach=None, state="active", started_at=0.0, finished=False,
                coaching=None):
    """Render one bounded synthetic fixture through the real render path."""
    practice = practice or {}
    coach = coach or {}
    relay, socket, handler = make_relay(f"probe/{label}")
    renderer = Renderer(relay)
    # Instance-level isolation, exactly as game_worker.setup() does it.
    renderer._used_group_ids = set()
    renderer._current_renders = []
    engine = FakeEngine(lesson, preferences, state=state, started_at=started_at,
                        finished=finished, coaching=coaching)
    session = build_session(
        lesson, preferences, native_hud=native_hud,
        practice=practice, coach=coach, engine=engine,
    )
    session.renderer = renderer
    # Verify the explicit seed list is present as real instance fields (not class
    # descriptors) before rendering. This checks only the attributes this probe
    # seeds; it cannot detect a new attribute a future render() might read.
    seeded = session_attributes_present(session)
    check(f"{label}:session_attributes_seeded",
          all(present for present, _ in seeded.values()), seeded)
    session.render(SYNTHETIC_FRAME)
    frames = decode_frames(socket)
    described = [describe_payload(frame["payload"]) for frame in frames]
    first = described[0] if described else None
    practice_visible = bool(practice.get("visible"))
    coach_visible = bool(coach.get("visible"))
    result = {
        "label": label,
        "lesson_id": lesson["id"],
        "native_hud": native_hud,
        "preferences": dict(preferences),
        "engine_state": state,
        "engine_started_at": started_at,
        "engine_finished": finished,
        "engine_summary": engine.summary(),
        "target_types": [target.get("type") for target in lesson["targets"]],
        "step_count": len(lesson["steps"]),
        "practice_visible": practice_visible,
        "coach_visible": coach_visible,
        "session_attributes_seeded": seeded,
        "frames_sent": len(frames),
        "payload_bytes": [frame["payload_bytes"] for frame in frames],
        "prefix_hex": [frame["prefix_hex"] for frame in frames],
        "prefix_matches_payload": bool(frames)
        and all(f["prefix_matches_payload"] for f in frames),
        "counts_available": bool(first) and first["kind"] == "RenderGroup",
        "message_counts": first["message_counts"] if first else {},
        "total_messages": first["total_messages"] if first else 0,
        "text_utf8_bytes": first["text_utf8_bytes"] if first else 0,
        "polyline_points": first["polyline_points"] if first else 0,
        "group_id": first["group_id"] if first else None,
        "expected_target_lines": expected_target_lines(lesson["targets"]),
        "expected_string_count": expected_string_count(
            steps=len(lesson["steps"]), preferences=preferences,
            native_hud=native_hud, practice_visible=practice_visible,
            coach_visible=coach_visible, engine_state=state,
            coaching=coaching or {},
        ),
        "logger_records": list(handler.records),
        "socket_connect_calls": socket.connect_calls,
        "socket_recv_calls": socket.recv_calls,
        "relay_running": relay._running,
        "renderer_queue_empty_after": len(renderer._current_renders) == 0,
        "renderer_group_id_cleared": renderer._group_id is None,
        "renderer_send_msg_is_production": (
            getattr(renderer._send_msg, "__func__", None) is SocketRelay.send_msg
        ),
    }
    return result, renderer, relay, socket, handler


def find_lesson(lessons, lesson_id):
    for item in lessons:
        if item["id"] == lesson_id:
            return item
    return None


def emit(report: dict) -> int:
    failures = [entry for entry in CHECKS if not entry["ok"]]
    report["checks"] = CHECKS
    report["check_count"] = len(CHECKS)
    report["failure_count"] = len(failures)
    report["failures"] = [entry["name"] for entry in failures]
    print(json.dumps(report, indent=2, sort_keys=True))
    return 1 if failures else 0


def main() -> int:
    report: dict = {
        "tool": "research_render_packet_probe",
        "scope": "client pipeline only; not a server, game or persistence simulation",
        "engine_fixture": "FakeEngine (synthetic, not live, not full engine)",
        "session_fixture": (

            "thread are never run"

            "127 is a conservative sum of independent per-branch maxima from "
            "docs/render-message-budget-01.md; it is not an attainable maximum "
            "and is not asserted against any measurement below."
        ),

            "engine_feedback": SYNTHETIC_FEEDBACK,
            "engine_duration": SYNTHETIC_DURATION,
            "oversized_text_length": OVERSIZED_TEXT_LENGTH,
        },
    }

    # ---- flatbuffer API preconditions (instance fields, not class guesses) --
    probe_group = flat.RenderGroup([], 0)
    probe_message = flat.RenderMessage(flat.Line3D())
    # InterfacePacket.message is a union that includes RenderGroup but not
    # RenderMessage (docs/rlbot_flatbuffers.pyi.txt:6157), so the packet probe
    # must wrap the already-built RenderGroup.
    probe_packet = flat.InterfacePacket(probe_group)
    api = {
        "RenderGroup.render_messages": hasattr(probe_group, "render_messages"),
        "RenderGroup.renders_absent": not hasattr(probe_group, "renders"),
        "RenderMessage.variety": hasattr(probe_message, "variety"),
        "RenderMessage.item_absent": not hasattr(probe_message, "item"),
        "InterfacePacket.message": hasattr(probe_packet, "message"),
        "InterfacePacket.message_is_render_group": isinstance(
            probe_packet.message, flat.RenderGroup),
        "RemoveRenderGroup.id": hasattr(flat.RemoveRenderGroup(0), "id"),
        "String2D.text": hasattr(flat.String2D("x"), "text"),
        "PolyLine3D.points": hasattr(flat.PolyLine3D([]), "points"),
    }
    report["flatbuffer_api"] = api
    for name, ok in api.items():
        check(f"api:{name}", ok)

    # The package re-exports a ``catalog`` function, so verify the bindings used
    # below are the submodules themselves and not the shadowing functions.
    bindings = {
        "catalog_module": catalog_module,
        "controller_module": controller_module,
        "recordings_module": recordings_module,
    }
    report["module_bindings"] = {
        name: {"is_module": isinstance(value, types.ModuleType),
               "name": getattr(value, "__name__", None)}
        for name, value in bindings.items()
    }
    for name, value in bindings.items():
        check(f"binding:{name}:is_module", isinstance(value, types.ModuleType),
              getattr(value, "__name__", type(value).__name__))
    check("binding:catalog_module:not_shadowed_function",
          catalog_module is not catalog_module.catalog,
          report["module_bindings"]["catalog_module"])

    # ---- identity of the code under measurement -------------------------
    identity = {
        "GameSession.render": GameSession.render,
        "GameSession.draw_target": GameSession.draw_target,
        "GameSession.draw_ghost_actions": GameSession.draw_ghost_actions,
        "Renderer.context": Renderer.context,
        "Renderer.end_rendering": Renderer.end_rendering,
        "SocketRelay.send_msg": SocketRelay.send_msg,
        "SocketRelay.send_bytes": SocketRelay.send_bytes,
        "ghost_lines": recordings_module.ghost_lines,
        "catalog": catalog_module.catalog,
        "lesson": catalog_module.lesson,
        "controller_display": controller_module.controller_display,
    }
    report["method_identity"] = {
        name: {"module": fn.__module__, "qualname": fn.__qualname__}
        for name, fn in identity.items()
    }
    check("render_is_production_function",
          identity["GameSession.render"].__module__ == "my_career_training.game_worker"
          and identity["GameSession.render"].__qualname__ == "GameSession.render")
    check("draw_target_is_production_function",

        ],
    }
    # Observation only: the proposed soft ceiling is a client-protocol margin
    # from docs/render-message-budget-01.md, not a measured no-drop threshold.
    report["soft_ceiling_observation"] = {
        "soft_ceiling_bytes": SOFT_CEILING_BYTES,
        "fixtures_over_soft_ceiling": [
            fixture["label"] for fixture in fixtures
            if any(size > SOFT_CEILING_BYTES for size in fixture["payload_bytes"])
        ],
        "note": "observation only; exceeding it is not asserted as a failure and "
                "staying under it is not asserted as safe",
    }
    return emit(report)
